// Load Pay Schedule from JSON let payScheduleData = null; async function loadPaySchedule() { try { showStatus('Loading pay schedule...', 'info'); // Fetch the JSON file const response = await fetch('./Pay_Schedule_FY2026.json'); if (!response.ok) throw new Error('Failed to load pay schedule'); payScheduleData = await response.json(); // Get today's date const today = new Date(); // Find the current pay period const currentPeriod = findCurrentPayPeriod(today, payScheduleData); if (!currentPeriod) { showStatus('Could not determine current pay period', 'error'); return; } // Find the retro period (6 periods back) const retroPeriod = findRetroPeriod(currentPeriod.payPeriod, payScheduleData); // Populate the form fields populateFormFields(currentPeriod, retroPeriod); showStatus('Pay schedule loaded successfully!', 'success'); } catch (error) { console.error('Error loading pay schedule:', error); showStatus('Error loading pay schedule: ' + error.message, 'error'); } } function findCurrentPayPeriod(date, scheduleData) { // Find the pay period where today falls within or after the period for (let i = 0; i < scheduleData.payPeriods.length; i++) { const period = scheduleData.payPeriods[i]; const payDate = new Date(period.payDate); const periodFrom = new Date(period.periodFrom); const periodThru = new Date(period.periodThru); // If today is within the period or before the pay date but after period start if (date >= periodFrom && date <= payDate) { return period; } // Check if we're between this pay date and next period start if (i < scheduleData.payPeriods.length - 1) { const nextPeriod = scheduleData.payPeriods[i + 1]; const nextPeriodFrom = new Date(nextPeriod.periodFrom); if (date > payDate && date < nextPeriodFrom) { return nextPeriod; } } } // If we're past all periods, return the last one return scheduleData.payPeriods[scheduleData.payPeriods.length - 1]; } function findRetroPeriod(currentPayPeriodNumber, scheduleData) { // Find the period 6 periods back const retroPayPeriodNumber = Math.max(1, currentPayPeriodNumber - 6); const currentIndex = scheduleData.payPeriods.findIndex(p => p.payPeriod === currentPayPeriodNumber); const retroIndex = scheduleData.payPeriods.findIndex(p => p.payPeriod === retroPayPeriodNumber); if (retroIndex === -1 || currentIndex === -1) return null; return { start: scheduleData.payPeriods[retroIndex].periodFrom, end: scheduleData.payPeriods[currentIndex - 1]?.periodThru || scheduleData.payPeriods[currentIndex].periodFrom }; } function populateFormFields(currentPeriod, retroPeriod) { // Populate Pay Date document.getElementById('payDateInput').value = currentPeriod.payDate; // Populate Current Period fields const periodFromDate = new Date(currentPeriod.periodFrom); const periodThruDate = new Date(currentPeriod.periodThru); const currentPeriodText = `${periodFromDate.toLocaleDateString('en-US', { month: 'long', day: 'numeric' })} - ${periodThruDate.toLocaleDateString('en-US', { month: 'long', day: 'numeric', year: 'numeric' })}`; document.getElementById('currentPeriod').value = currentPeriodText; // Populate Current Due dates (convert ISO to datetime-local format) const employeeDueDate = new Date(currentPeriod.currentDue.employee); const supervisorDueDate = new Date(currentPeriod.currentDue.supervisor); document.getElementById('currentDueWorkerDateTime').value = formatDateTimeLocal(employeeDueDate); document.getElementById('currentDueManagerDateTime').value = formatDateTimeLocal(supervisorDueDate); // Populate Retro Period fields if (retroPeriod) { const retroStartDate = new Date(retroPeriod.start); const retroEndDate = new Date(retroPeriod.end); const retroPeriodText = `${retroStartDate.toLocaleDateString('en-US', { month: 'long', day: 'numeric' })} - ${retroEndDate.toLocaleDateString('en-US', { month: 'long', day: 'numeric', year: 'numeric' })}`; document.getElementById('retroPeriod').value = retroPeriodText; } // Populate Retro Due dates const retroEmployeeDueDate = new Date(currentPeriod.retroDue.employee); const retroSupervisorDueDate = new Date(currentPeriod.retroDue.supervisor); document.getElementById('retroDueWorkerDateTime').value = formatDateTimeLocal(retroEmployeeDueDate); document.getElementById('retroDueManagerDateTime').value = formatDateTimeLocal(retroSupervisorDueDate); } function formatDateTimeLocal(date) { // Format date as YYYY-MM-DDTHH:MM for datetime-local input const year = date.getFullYear(); const month = String(date.getMonth() + 1).padStart(2, '0'); const day = String(date.getDate()).padStart(2, '0'); const hours = String(date.getHours()).padStart(2, '0'); const minutes = String(date.getMinutes()).padStart(2, '0'); return `${year}-${month}-${day}T${hours}:${minutes}`; }
